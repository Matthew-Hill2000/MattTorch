109
110 for (int i{0}; i < static_cast<int>(layers.size()); ++i) {

112 // Using RTTI to determine layer type
113 const auto& layer = layers[i];
114 if (dynamic_cast<FullyConnectedLayer*>(layer)) {
115 ss << "Fully Connected Layer";

117 ss << "ReLU Activation";

119 ss << "Tanh Activation";
120 } else {
121 ss << "Unknown Layer Type";
122 }
123 ss << std::endl;
124 }
