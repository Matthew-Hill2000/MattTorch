29TensorView::TensorView(const std::vector<int>& dims, bool isLeaf)
30 : gradient{nullptr},
31 isLeaf{isLeaf},
32 requiresGrad{true},
33 hasGrad{true},
34 dimensions{dims},
35 strides{},
36 offset{0},
37 nValues{1},
38 rank{static_cast<int>(dims.size())} {
39 strides.resize(rank);
40
41 if (rank == 0) {
42 nValues = 0;
43 } else {
44 strides[rank - 1] = 1;
45 nValues = dimensions[rank - 1];
46 for (int i{rank - 2}; i >= 0; i--) {
47 strides[i] = strides[i + 1] * dimensions[i + 1];
48 nValues *= dimensions[i];
49 }
50 }
51
52 storage = std::make_shared<tensor::TensorStorage>(nValues);
53 if (isLeaf) {
54 gradient = std::make_shared<TensorView>(dimensions, false);
55 gradFunction =
56 std::make_shared<function::GradAccumulator>(gradient, dimensions);
57 }
58}

92TensorView TensorView::deepCopy() const {
93 TensorView result(this->dimensions);
94
95 double* rhs = this->getData();
96 double* lhs = result.getData();
97
98 int i{0};
99 for (; i + 3 < nValues; i += 4) {
100 __m256d a = _mm256_loadu_pd(rhs + i);
101 _mm256_storeu_pd(lhs + i, a);
102 }
103
104 for (; i < nValues; i++) {
105 lhs[i] = rhs[i];
106 }
107
108 return result;
109}
