62int TensorView::calculateIndex(const std::vector<int>& indices) const {
63 if (static_cast<int>(indices.size()) != rank) {
64 throw std::invalid_argument("Number of indices doesn't match tensor rank");
65 }
66
67 for (int i{0}; i < static_cast<int>(indices.size()); i++) {
68 if (indices[i] < 0 || indices[i] >= dimensions[i]) {
69 throw std::out_of_range("Index out of bounds for dimension " +
70 std::to_string(i));
71 }
72 }
73
74 int index{offset};
75 for (int i{0}; i < rank; i++) {
76 index += indices[i] * strides[i];
77 }
78
79 return index;
80}
