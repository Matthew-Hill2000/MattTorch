88void dataset::shuffle() {
89 if (indices.size() < 2) return;
90
91 std::random_device rd;
92 std::mt19937 gen(rd());
93
94 for (size_t i = indices.size() - 1; i > 0; --i) {
95 std::uniform_int_distribution<size_t> dist(0, i);
96 size_t j = dist(gen);
97 std::swap(indices[i], indices[j]);
98 }
