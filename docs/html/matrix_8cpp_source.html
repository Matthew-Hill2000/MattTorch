47void matrixMultiplicationInverseVector(const double* __restrict lhs,
48 const double* __restrict rhs,
49 double* __restrict result, int lhsRows,
50 int lhsCols, int rhsCols) {
51 double* rhsTranspose = nullptr;
52 posix_memalign(reinterpret_cast<void**>(&rhsTranspose), 32,
53 lhsCols * rhsCols * sizeof(double));
54
55 for (int i = 0; i < lhsCols; i++) {
56 for (int j = 0; j < rhsCols; j++) {
57 rhsTranspose[lhsCols * j + i] = rhs[rhsCols * i + j];
58 }
59 }
60
61 for (int i = 0; i < lhsRows; i++) {
62 for (int j = 0; j < rhsCols; j++) {
63 __m256d acc = _mm256_setzero_pd();
64
65 int k = 0;
66 for (; k + 3 < lhsCols; k += 4) {
67 __m256d a = _mm256_loadu_pd(&lhs[lhsCols * i + k]);
68 __m256d b = _mm256_loadu_pd(&rhsTranspose[lhsCols * j + k]);
69 acc = _mm256_fmadd_pd(a, b, acc);
70 }
71
72 __m128d lo = _mm256_castpd256_pd128(acc);
73 __m128d hi = _mm256_extractf128_pd(acc, 1);
74 __m128d sum = _mm_add_pd(lo, hi);
75 sum = _mm_hadd_pd(sum, sum);
76
77 double result_ij = _mm_cvtsd_f64(sum);
78
79 for (; k < lhsCols; k++) {
80 result_ij += lhs[lhsCols * i + k] * rhsTranspose[lhsCols * j + k];
81 }
82
83 result[rhsCols * i + j] = result_ij;
84 }
85 }
86
87 free(rhsTranspose);

90void matrixMultiplicationBlock(const double* __restrict lhs,
91 const double* __restrict rhs,
92 double* __restrict result, int lhsRows,
93 int lhsCols, int rhsCols) {
94 int blockSize = 32;
95
96 double* rhsBlockTranspose = nullptr;
97 posix_memalign(reinterpret_cast<void**>(&rhsBlockTranspose), 32,
98 blockSize * blockSize * sizeof(double));
99
100 // For each Block of rhs along a Row
101 int j{0};
102 for (; rhsCols - j >= blockSize; j += blockSize) {
103 // For each Block of rhs along a column and lhs across a row
104 int k{0};
105 for (; lhsCols - k >= blockSize; k += blockSize) {
106 // Take the transpose of the block of rhs
107 // For each Row of the rhs block
108 for (int x{0}; x < blockSize; x++) {
109 // For each column of the rhs block
110 for (int y{0}; y < blockSize; y++) {
111 rhsBlockTranspose[y * blockSize + x] = rhs[(k + x) * rhsCols + j + y];
112 }
113 }
114
115 // Perform the matrix multiplication of the lhs and rhs blocks
116 // For each Block of lhs along a Column
117 int i{0};
118 for (; lhsRows - i >= blockSize; i += blockSize) {
119 // For each row along the lhs block
120 for (int x{0}; x < blockSize; x++) {
121 // For each column along the rhs block
122 for (int y{0}; y < blockSize; y++) {
123 // For each column along the lhs block and row along the rhs block
124 for (int z{0}; z < blockSize; z++) {
125 result[(i + x) * rhsCols + j + y] +=
126 lhs[(i + x) * lhsCols + k + z] *
127 rhsBlockTranspose[y * blockSize + z];
128 }
129 }
130 }
131 }
132
133 // When you reach the last entire block down a column of lhs, you then
134 // need to multiply the rhs block by the remainder block at the bottom of
135 // the lhs column
136 // For each row remaining in ths column
137 for (; i < lhsRows; i++) {
138 // for each column of the rhs block
139 for (int y{0}; y < blockSize; y++) {
140 // For each column of the lhs remainder block and row of the rhs block
141 for (int z{0}; z < blockSize; z++) {
142 result[i * rhsCols + j + y] +=
143 lhs[i * lhsCols + k + z] * rhsBlockTranspose[y * blockSize + z];
144 }
145 }
146 }
147 }
148
149 // When you reach the last entire block down a column of the rhs, you then
150 // need to multiply the remainder block at the bottom of the rhs column by
151 // the remainder block at the end of every row of the lhs
152 // For each column/row reamaining in the lhs/rhs remainder block
153 for (; k < lhsCols; k++) {
154 // For each remainder block in the lhs
155 int i{0};
156 for (; lhsRows - i >= blockSize; i += blockSize) {
157 // For each row of the remainder block of the lhs
158 for (int x{0}; x < blockSize; x++) {
159 // For each column of the remainder block of the rhs
160 for (int y{0}; y < blockSize; y++) {
161 result[((i + x) * rhsCols) + j + y] +=
162 lhs[(i + x) * lhsCols + k] * rhs[k * rhsCols + j + y];
163 }
164 }
165 }
166 for (; i < lhsRows; i++) {
167 for (int y{0}; y < blockSize; y++) {
168 result[i * rhsCols + j + y] +=
169 lhs[i * lhsCols + k] * rhs[k * rhsCols + j + y];
170 }
171 }
172 }
173 }
174 // When you reach the last entire column of blocks in the rhs, you then need
175 // to multiply the column of remainder blocks by each corresponding block
176 // across each row of blocks of the lhs
177 // For each column of the remaining rhs remainder block
178 for (; j < rhsCols; j++) {
179 // For each remainder block down the last column of rhs and each block
180 // across a row of lhs
181 int k{0};
182 for (; lhsCols - k >= blockSize; k += blockSize) {
183 // For each block of lhs down a column
184 int i{0};
185 for (; lhsRows - i >= blockSize; i += blockSize) {
186 // For each row of the lhs block
187 for (int x{0}; x < blockSize; x++) {
188 // For each column of the lhs block and each row of the rhs remainder
189 // block
190 for (int z{0}; z < blockSize; z++) {
191 result[(i + x) * rhsCols + j] +=
192 lhs[(i + x) * lhsCols + k + z] * rhs[(k + z) * rhsCols + j];
193 }
194 }
195 }
196 // when you reach the last entire row of blocks, multiply by the bottom
197 // right hand residual block
198 for (; i < lhsRows; i++) {
199 for (int z{0}; z < blockSize; z++) {
200 result[i * rhsCols + j] +=
201 lhs[i * lhsCols + k + z] * rhs[(k + z) * rhsCols + j];
202 }
203 }
204 }
205 // When you reach the last remainder block of the rhs column of remainder
206 // blocks of the rhs, you need to then multiply the small bottom right hand
207 // corner remainder block of the rhs by each of the remainder blocks at the
208 // end of each row of lhs
209 // For each row of the little remainder block and column of the lhs
210 // remainder blocks
211 for (; k < lhsCols; k++) {
212 // For each remainder block down the lhs column of remainder blocks.
213 int i{0};
214 for (; lhsRows - i >= blockSize; i += blockSize) {
215 // For each row of the lhs remainder block
216 for (int x{0}; x < blockSize; x++) {
217 result[(i + x) * rhsCols + j] +=
218 lhs[(i + x) * lhsCols + k] * rhs[k * rhsCols + j];
219 }
220 }
221 // When you reach the end of the lhs column of remainders
222 // Finally multiply the tiny remainder block of lhs bottom corner and
223 // the
224 // tiny remainder block of rhs bottom corner
225 for (; i < lhsRows; i++) {
226 result[i * rhsCols + j] += lhs[i * lhsCols + k] * rhs[k * rhsCols + j];
227 }
228 }
229 free(rhsBlockTranspose);
230 }
